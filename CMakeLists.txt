cmake_minimum_required(VERSION 3.20)
project(YoruTokyo VERSION 1.26.0 LANGUAGES C CXX)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(YORU_BUILD_APP "Build the native city builder" ON)
option(YORU_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(yoru_city src/BuildCatalog.cpp src/ExpandedDemo.cpp src/City.cpp src/Persistence.cpp src/Production.cpp src/Simulation.cpp src/Spatial.cpp src/Infrastructure.cpp src/Construction.cpp src/Demo.cpp src/RoutePlanner.cpp src/MachineTrace.cpp src/Districts.cpp src/DistrictTrace.cpp src/Transport.cpp)
target_include_directories(yoru_city PUBLIC src vendor)
target_compile_options(yoru_city PRIVATE -Wall -Wextra -Wpedantic)
if(YORU_SANITIZE)
  target_compile_options(yoru_city PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(yoru_city PUBLIC -fsanitize=address,undefined)
endif()
if(YORU_BUILD_APP)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  add_subdirectory(vendor/raylib EXCLUDE_FROM_ALL)
  target_compile_definitions(raylib PRIVATE YORU_STREAM_BATCH=1)
  if(APPLE)
    add_executable(YoruTokyo MACOSX_BUNDLE src/Main.cpp src/EasyBuildUI.cpp src/Interface.cpp src/CityRendering.cpp src/CityMachine.cpp src/OpticsNavigationCheck.cpp src/SimulationWorker.cpp src/Schematic.cpp src/Kernel.cpp src/SpatialUI.cpp src/EcologyUI.cpp src/ResilienceUI.cpp src/ConstructionUI.cpp src/DemoUI.cpp src/RenderBenchmark.cpp src/RenderChecks.cpp src/FlowUpdateCheck.cpp src/RenderResources.cpp src/RenderAssets.cpp src/MinimalUI.cpp src/RouteUI.cpp src/TraceUI.cpp src/RouteTraceCheck.cpp src/DistrictUI.cpp src/DistrictCheck.cpp src/TransportUI.cpp src/TransportCheck.cpp)
    set_target_properties(YoruTokyo PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Yoru Tokyo"
      MACOSX_BUNDLE_GUI_IDENTIFIER "studio.yoru.tokyo"
      MACOSX_BUNDLE_BUNDLE_VERSION "1.26.0" MACOSX_BUNDLE_SHORT_VERSION_STRING "1.26.0"
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
    add_custom_command(TARGET YoruTokyo POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_CONTENT_DIR:YoruTokyo>/Resources"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/README.txt" "$<TARGET_BUNDLE_CONTENT_DIR:YoruTokyo>/Resources/README.txt"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/vendor/raylib/LICENSE" "$<TARGET_BUNDLE_CONTENT_DIR:YoruTokyo>/Resources/raylib-LICENSE.txt"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/vendor/json-LICENSE.txt" "$<TARGET_BUNDLE_CONTENT_DIR:YoruTokyo>/Resources/json-LICENSE.txt"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/RENDER-PERFORMANCE.txt" "$<TARGET_BUNDLE_CONTENT_DIR:YoruTokyo>/Resources/RENDER-PERFORMANCE.txt"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/ORIGIN.txt" "$<TARGET_BUNDLE_CONTENT_DIR:YoruTokyo>/Resources/ORIGIN.txt"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/VALIDATION.txt" "$<TARGET_BUNDLE_CONTENT_DIR:YoruTokyo>/Resources/VALIDATION.txt"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/VERTICALITY-RESEARCH.txt" "$<TARGET_BUNDLE_CONTENT_DIR:YoruTokyo>/Resources/VERTICALITY-RESEARCH.txt")
  else()
    add_executable(YoruTokyo src/Main.cpp src/EasyBuildUI.cpp src/Interface.cpp src/CityRendering.cpp src/CityMachine.cpp src/OpticsNavigationCheck.cpp src/SimulationWorker.cpp src/Schematic.cpp src/Kernel.cpp src/SpatialUI.cpp src/EcologyUI.cpp src/ResilienceUI.cpp src/ConstructionUI.cpp src/DemoUI.cpp src/RenderBenchmark.cpp src/RenderChecks.cpp src/FlowUpdateCheck.cpp src/RenderResources.cpp src/RenderAssets.cpp src/MinimalUI.cpp src/RouteUI.cpp src/TraceUI.cpp src/RouteTraceCheck.cpp src/DistrictUI.cpp src/DistrictCheck.cpp src/TransportUI.cpp src/TransportCheck.cpp)
  endif()
  target_link_libraries(YoruTokyo PRIVATE yoru_city raylib Threads::Threads)
  target_compile_options(YoruTokyo PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS YoruTokyo BUNDLE DESTINATION . RUNTIME DESTINATION bin)
endif()
